stand3 <- function(data) {
  # 1. Access the parameters table from the summary
  params <- data$sum$parameters
  
  # 2. Pull the 'Estimate' value where the 'name' matches your labels
  # We use [1] at the end just to ensure we get a single numeric value
  A <- params$Estimate[params$name == "VA11"]
  C <- params$Estimate[params$name == "VC11"]
  E <- params$Estimate[params$name == "VE11"]
  
  # 3. Calculation
  V  <- (A + C + E)
  a2 <- A / V
  c2 <- C / V
  e2 <- E / V
  
  # 4. Return as a clean row
  estACE <- rbind(cbind(a2,c2,e2))
  return(estACE)
}
