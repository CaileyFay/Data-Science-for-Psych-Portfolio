for (i in 1:length(df_list)) {
  
  df_list[[i]] <- df_list[[i]] %>%

 mutate(y1 = case_when(
    R == .5 ~ ifelse(runif(n()) < 0.20, NA, y1), 
     R == .25 ~ ifelse(runif(n()) < 0.10, NA, y1)))  %>%
    
  mutate(y2 = case_when(
